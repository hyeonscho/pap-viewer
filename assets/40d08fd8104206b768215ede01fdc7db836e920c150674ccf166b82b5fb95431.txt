def _prepare(observation):
    a = np.asarray(observation)
    if a.shape != (224, 224) or a.dtype != np.uint8:
        raise ValueError('Expected a (224,224) uint8 array')
    yy, xx = np.nonzero(a != 128)
    if not len(xx):
        return None
    # Possible window origins form a Cartesian product of integer intervals.
    ox0, ox1 = max(0, int(xx.max()) - 63), min(160, int(xx.min()))
    oy0, oy1 = max(0, int(yy.max()) - 63), min(160, int(yy.min()))
    if ox0 > ox1 or oy0 > oy1:
        raise ValueError('No consistent in-bounds 64x64 visibility window')
    x0, x1 = ox1, ox0 + 64
    y0, y1 = oy1, oy0 + 64
    a = a[y0:y1, x0:x1]
    # A genuine visible 128 remains unknown photometric evidence, not background.
    dark, light = a <= 96, a >= 176
    labels, _ = ndimage.label(dark, structure=np.ones((3, 3), dtype=bool))
    return a, dark, light, labels, x0, y0


def _instance(p, concept, anchor, box):
    _, _, _, _, ox, oy = p
    x, y = anchor
    x0, y0, x1, y1 = box
    return {'concept_id': concept,
            'anchor': [int(np.rint(x)) + ox, int(np.rint(y)) + oy],
            'box': [int(x0) + ox, int(y0) + oy,
                    int(x1) + ox, int(y1) + oy]}


def _runs(bits):
    v = np.flatnonzero(np.diff(np.r_[False, bits, False].astype(np.int8)))
    return zip(v[::2], v[1::2])


def _sections(p, axis, margin=2):
    _, d, l, labels, _, _ = p
    if axis == 'x':
        d, l, labels = d.T, l.T, labels.T
    rows = []
    for t in range(d.shape[0]):
        nodes = []
        for lo, end in _runs(d[t]):
            if end - lo < 4 or end - lo > 48:
                continue
            if lo < margin or end + margin > d.shape[1]:
                continue
            if not (l[t, lo-margin:lo].all() and l[t, end:end+margin].all()):
                continue
            nodes.append((t, float(lo), float(end - 1), int(labels[t, lo])))
        rows.append(nodes)
    return rows


def _chains(rows, jump=6.0):
    active, finished = [], []
    for nodes in rows:
        pairs = []
        for i, chain in enumerate(active):
            old = chain[-1]
            for j, node in enumerate(nodes):
                if old[3] != node[3]:
                    continue
                delta = abs((old[1] + old[2] - node[1] - node[2]) / 2)
                if delta <= jump:
                    pairs.append((delta, i, j))
        used_i, used_j, following = set(), set(), []
        for _, i, j in sorted(pairs):
            if i not in used_i and j not in used_j:
                following.append(active[i] + [nodes[j]])
                used_i.add(i)
                used_j.add(j)
        finished.extend(c for i, c in enumerate(active) if i not in used_i)
        following.extend([n] for j, n in enumerate(nodes) if j not in used_j)
        active = following
    return finished + active


def _line(t, q):
    tc, qc = float(np.mean(t)), float(np.mean(q))
    den = float(np.sum((t - tc) ** 2))
    m = float(np.sum((t - tc) * (q - qc)) / den) if den else 0.0
    b = qc - m * tc
    error = float(np.max(np.abs(q - (m * t + b))))
    return m, b, error


def _bands(observation, kind):
    p = _prepare(observation)
    if p is None:
        return []
    axes = ('y', 'x') if kind == 2 else (('y',) if kind == 0 else ('x',))
    best = {}
    for axis in axes:
        for chain in _chains(_sections(p, axis)):
            if len(chain) < 12:
                continue
            z = np.asarray(chain, dtype=float)
            widths = z[:, 2] - z[:, 1] + 1
            reference = float(np.quantile(widths, 0.75))
            for start in range(len(z) - 11):
                for end in range(len(z), start + 11, -1):
                    s = z[start:end]
                    t, lo, hi = s[:, 0], s[:, 1], s[:, 2]
                    w = hi - lo + 1
                    if np.min(w) < (0.65 if kind == 2 else 0.80) * reference:
                        continue
                    center = (lo + hi) / 2
                    m, b, err = _line(t, center)
                    if kind == 2:
                        if not (-3.2 <= m <= -0.32) or abs(center[-1] - center[0]) < 7:
                            continue
                    elif abs(m) > 0.32:
                        continue
                    ml, _, el = _line(t, lo)
                    mh, _, eh = _line(t, hi)
                    tolerance = 0.42 if kind == 2 else 0.23
                    if abs(ml - mh) > tolerance:
                        continue
                    if max(err, el, eh) > (3.0 if kind == 2 else 2.1):
                        continue
                    if np.ptp(w) > max(4.0, (0.38 if kind == 2 else 0.24) * np.median(w)):
                        continue
                    # A taper cannot provide the parallel-contour evidence.
                    if abs(np.median(w[:3]) - np.median(w[-3:])) > max(2.5, 0.18 * np.median(w)):
                        continue
                    if np.any(np.maximum(lo[1:], lo[:-1]) > np.minimum(hi[1:], hi[:-1]) + 1):
                        continue
                    length = float((t[-1] - t[0]) * np.sqrt(1 + m*m))
                    if length < 14:
                        continue
                    mid = (t[0] + t[-1]) / 2
                    transverse = m * mid + b
                    if axis == 'y':
                        anchor = (transverse, mid)
                        box = (int(lo.min()) - 2, int(t[0]), int(hi.max()) + 3, int(t[-1]) + 1)
                    else:
                        anchor = (mid, transverse)
                        box = (int(t[0]), int(lo.min()) - 2, int(t[-1]) + 1, int(hi.max()) + 3)
                    key = int(s[0, 3])
                    score = length - err
                    if key not in best or score > best[key][0]:
                        best[key] = (score, _instance(p, _IDS[kind], anchor, box))
                    break
    return [best[k][1] for k in sorted(best)]

def _edge_parts(p, axis, dark_positive):
    _, d, l, labels, _, _ = p
    if axis == 'h':
        d, l, labels = d.T, l.T, labels.T
    rows = []
    for t in range(d.shape[0]):
        if dark_positive:
            q = np.flatnonzero(l[t, :-1] & d[t, 1:])
            nodes = [(t, float(v) + 0.5, float(v) + 0.5, int(labels[t, v+1])) for v in q]
        else:
            q = np.flatnonzero(d[t, :-1] & l[t, 1:])
            nodes = [(t, float(v) + 0.5, float(v) + 0.5, int(labels[t, v])) for v in q]
        rows.append(nodes)
    parts = []
    for chain in _chains(rows, jump=3):
        if len(chain) < 9:
            continue
        z = np.asarray(chain, dtype=float)
        for size in (9, 13, 19, 27, 39, 55):
            if size > len(z):
                continue
            starts = sorted(set(list(range(0, len(z)-size+1, 3)) + [len(z)-size]))
            for start in starts:
                s = z[start:start+size]
                m, b, err = _line(s[:, 0], s[:, 1])
                if abs(m) <= 0.32 and err <= 1.4:
                    parts.append((m, b, s, int(s[0, 3])))
    return parts


def _arm(p, part, axis, positive):
    _, d, l, _, _, _ = p
    points, support, widths = [], [], []
    for t, q, _, _ in part[2]:
        t = int(t)
        k = int(np.ceil(q)) if positive else int(np.floor(q))
        profile = d[:, t] if axis == 'h' else d[t, :]
        lights = l[:, t] if axis == 'h' else l[t, :]
        if not (0 <= k < len(profile)) or not profile[k]:
            continue
        lo, hi = k, k
        while lo > 0 and profile[lo-1]:
            lo -= 1
        while hi+1 < len(profile) and profile[hi+1]:
            hi += 1
        if lo == 0 or hi+1 == len(profile) or not (lights[lo-1] and lights[hi+1]):
            continue
        width = hi-lo+1
        if not 5 <= width <= 46:
            continue
        points.append((t, (lo+hi)/2))
        widths.append(width)
        if axis == 'h':
            support.extend(((t, lo-1), (t, hi+1)))
        else:
            support.extend(((lo-1, t), (hi+1, t)))
    if len(points) < 5:
        return None
    pts = np.asarray(points, dtype=float)
    m, b, err = _line(pts[:, 0], pts[:, 1])
    if abs(m) > 0.4 or err > 2.5:
        return None
    return m, b, float(np.median(widths)), support


def _supported_interior_arm(p, part, axis, positive, width):
    # A cropped outer contour is not fitted. Instead, a measured transverse
    # thickness from the other arm defines an entirely visible interior strip.
    _, d, _, _, _, _ = p
    sign = 1 if positive else -1
    support, centers = [], []
    for t, q, _, _ in part[2]:
        t = int(t)
        near = int(np.ceil(q)) if positive else int(np.floor(q))
        far = near + sign * (int(np.rint(width))-1)
        lo, hi = min(near, far), max(near, far)
        profile = d[:, t] if axis == 'h' else d[t, :]
        if lo < 0 or hi >= len(profile) or not profile[lo:hi+1].all():
            return None
        centers.append((t, (lo+hi)/2))
        if axis == 'h':
            support.extend(((t, lo), (t, hi)))
        else:
            support.extend(((lo, t), (hi, t)))
    pts = np.asarray(centers, dtype=float)
    m, b, err = _line(pts[:, 0], pts[:, 1])
    if err > 2.5:
        return None
    return m, b, width, support


def _elbows(observation, left_down):
    p = _prepare(observation)
    if p is None:
        return []
    _, d, light, _, _, _ = p
    # Horizontal/vertical interfaces bordering the specified light recess.
    hs = _edge_parts(p, 'h', not left_down)
    vs = _edge_parts(p, 'v', left_down)
    best = {}
    arm_cache = {}
    def arm(part, axis, positive):
        key = (id(part), axis, positive)
        if key not in arm_cache:
            arm_cache[key] = _arm(p, part, axis, positive)
        return arm_cache[key]
    for h in hs:
        mh, bh, sh, component = h
        for v in vs:
            mv, bv, sv, vc = v
            if component != vc:
                continue
            ix = (mv*bh+bv)/(1-mv*mh)
            iy = mh*ix+bh
            if left_down:
                valid = (sh[0, 0] <= ix-8 and -3 <= ix-sh[-1, 0] <= 13
                         and sv[-1, 0] >= iy+8 and -3 <= sv[0, 0]-iy <= 13)
                rx, ry = ix-4, iy+4
            else:
                valid = (sh[-1, 0] >= ix+8 and -3 <= sh[0, 0]-ix <= 13
                         and sv[0, 0] <= iy-8 and -3 <= iy-sv[-1, 0] <= 13)
                rx, ry = ix+4, iy-4
            if not valid:
                continue
            rx, ry = int(np.rint(rx)), int(np.rint(ry))
            if not (1 <= rx < d.shape[1]-1 and 1 <= ry < d.shape[0]-1):
                continue
            if not light[ry-1:ry+2, rx-1:rx+2].all():
                continue
            ah = arm(h, 'h', not left_down)
            av = arm(v, 'v', left_down)
            if left_down:
                if ah is None and av is not None:
                    ah = _supported_interior_arm(p, h, 'h', False, av[2])
                if av is None and ah is not None:
                    av = _supported_interior_arm(p, v, 'v', True, ah[2])
            if ah is None or av is None:
                continue
            if max(ah[2], av[2]) > 1.8*min(ah[2], av[2]):
                continue
            hm, hb, _, hp = ah
            vm, vb, _, vp = av
            ax = (vm*hb+vb)/(1-vm*hm)
            ay = hm*ax+hb
            if not (0 <= ax < d.shape[1] and 0 <= ay < d.shape[0]):
                continue
            samples = list(hp)+list(vp)+[(rx-1, ry-1), (rx+1, ry+1)]
            # Include each dark/light edge pair, including rounded-transition
            # gaps between the two fitted inner-contour branches.
            for t, q, _, _ in sh:
                samples.extend(((int(t), int(np.floor(q))), (int(t), int(np.ceil(q)))))
            for t, q, _, _ in sv:
                samples.extend(((int(np.floor(q)), int(t)), (int(np.ceil(q)), int(t))))
            ht = float(np.mean(sh[:, 0]))
            vt = float(np.mean(sv[:, 0]))
            endpoints = [(ht, hm*ht+hb), (vm*vt+vb, vt), (ax, ay)]
            samples.extend(endpoints)
            xy = np.asarray(samples, dtype=float)
            x0, y0 = np.floor(xy.min(axis=0)).astype(int)
            x1, y1 = np.ceil(xy.max(axis=0)).astype(int)+1
            if x0 < 0 or y0 < 0 or x1 > d.shape[1] or y1 > d.shape[0]:
                continue
            local, _ = ndimage.label(d[y0:y1, x0:x1], structure=np.ones((3, 3), bool))
            tags = []
            for x, y in endpoints:
                xx, yy = int(np.rint(x)), int(np.rint(y))
                if not (x0 <= xx < x1 and y0 <= yy < y1):
                    tags.append(0)
                else:
                    tags.append(int(local[yy-y0, xx-x0]))
            if not tags[0] or len(set(tags)) != 1:
                continue
            score = len(sh)+len(sv)
            if component not in best or score > best[component][0]:
                concept = _IDS[3 if left_down else 4]
                best[component] = (score, _instance(p, concept, (ax, ay), (x0, y0, x1, y1)))
    return [best[k][1] for k in sorted(best)]

def detect_batch0_rounded_downward_tip(observation):
    p = _prepare(observation)
    if p is None:
        return []
    _, d, light, labels, _, _ = p
    result = []
    for component in range(1, int(labels.max())+1):
        yy, xx = np.nonzero(labels == component)
        bottom = int(yy.max())
        if bottom+1 >= d.shape[0] or bottom < 12:
            continue
        terminal = xx[yy == bottom]
        tx = float((terminal.min()+terminal.max())/2)
        # Locate a short shaft above the terminal narrowing.
        probes = []
        for y in range(max(0, bottom-15), bottom-5):
            for lo, end in _runs(labels[y] == component):
                if lo <= tx <= end-1 and 6 <= end-lo <= 42:
                    probes.append(end-lo)
        if len(probes) < 4:
            continue
        width = float(np.median(probes))
        cap_height = max(4, int(np.ceil(0.48*width)))
        start = bottom-cap_height-7
        if start < 0:
            continue
        rows = []
        good = True
        for y in range(start, bottom+1):
            runs = [(lo, end) for lo, end in _runs(labels[y] == component)
                    if lo <= tx+width and end-1 >= tx-width]
            if len(runs) != 1:
                good = False
                break
            lo, end = runs[0]
            if lo < 1 or end >= d.shape[1] or not (light[y, lo-1] and light[y, end]):
                good = False
                break
            rows.append((y, lo, end-1))
        if not good:
            continue
        z = np.asarray(rows, dtype=float)
        widths = z[:, 2]-z[:, 1]+1
        shaft = z[:7]
        sm, _, se = _line(shaft[:, 0], (shaft[:, 1]+shaft[:, 2])/2)
        if abs(sm) > 0.65 or se > 1.8:
            continue
        sw = float(np.median(widths[:7]))
        if np.ptp(widths[:7]) > max(4, 0.25*sw):
            continue
        if widths[-1] > 0.75*sw or widths[-1] < 1:
            continue
        # Both sides must turn inward, rather than a single diagonal cut.
        predicted_shift = sm*(bottom-float(np.mean(shaft[:, 0])))
        left0 = float(np.mean(shaft[:, 1]))+predicted_shift
        right0 = float(np.mean(shaft[:, 2]))+predicted_shift
        if z[-1, 1]-left0 < 2 or right0-z[-1, 2] < 2:
            continue
        if np.count_nonzero(np.diff(widths[-cap_height:]) > 2) > 1:
            continue
        lo, hi = int(z[:, 1].min()), int(z[:, 2].max())
        # A full light row below the cap supplies genuine termination evidence.
        if not light[bottom+1, lo-1:hi+2].all():
            continue
        anchor = ((z[-1, 1]+z[-1, 2])/2, bottom)
        box = (lo-1, start, hi+2, bottom+2)
        result.append(_instance(p, _IDS[5], anchor, box))
    return result
